Alternatively, you can also use this JavaScript snippet as a userscript or in the browser console:

const id = new URLSearchParams(window.location.search).get('v') || window.location.pathname.match(/^\/([\w-]{11})/)?.[1]; if (!id) { alert('Not on a YouTube video page.'); } else { fetch('http://localhost:7823/summary', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({video_id: id, save: true}) }).then(r => r.json()).then(d => { if (d.summary) alert('Summary for: ' + d.title + '\n\n' + d.summary); else alert('Error: ' + (d.error || 'Unknown')); }).catch(() => alert('Server not running on port 7823.')); }
